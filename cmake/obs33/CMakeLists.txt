cmake_minimum_required(VERSION 3.28)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../buildspec.json" SPEC)
string(JSON VERSION GET "${SPEC}" version)
project(scene-anchor VERSION ${VERSION} LANGUAGES C CXX)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets)
set(OBS_SOURCE_DIR "" CACHE PATH "Official OBS 33 source headers")
set(OBS_IMPORT_DIR "" CACHE PATH "OBS 33 import libraries")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/generated/obsconfig.h")
configure_file("${ROOT}/src/plugin-support.c.in" "${CMAKE_BINARY_DIR}/generated/plugin-support.c" @ONLY)
add_library(scene-anchor MODULE "${ROOT}/src/module.cpp" "${ROOT}/src/obs_bridge.cpp" "${ROOT}/src/obs_bridge.h" "${ROOT}/src/tree_dock.cpp" "${ROOT}/src/tree_dock.h" "${ROOT}/src/tree_store.cpp" "${ROOT}/src/projection.cpp" "${CMAKE_BINARY_DIR}/generated/plugin-support.c")
target_include_directories(scene-anchor PRIVATE "${ROOT}/src" "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(scene-anchor PRIVATE Qt6::Widgets "${OBS_IMPORT_DIR}/obs.lib" "${OBS_IMPORT_DIR}/obs-frontend-api.lib")
target_compile_options(scene-anchor PRIVATE /utf-8)
set_target_properties(scene-anchor PROPERTIES PREFIX "")
install(TARGETS scene-anchor RUNTIME DESTINATION plugins/scene-anchor LIBRARY DESTINATION plugins/scene-anchor)
install(DIRECTORY "${ROOT}/data/" DESTINATION plugins/scene-anchor/data)
